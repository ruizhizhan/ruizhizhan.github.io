{{- $photos := slice -}} {{- range site.Data.photography.photos -}} {{- $entry := . -}} {{- with resources.Get (printf "media/albums/%s" .src) -}} {{- $thumbnail := . -}} {{- if or (gt .Width 1000) (gt .Height 900) -}} {{- $thumbnail = .Fit "1000x900 webp q86" -}} {{- end -}} {{- $preview := .Fit "48x48 webp q55" -}} {{- $dateLabel := "Undated" -}} {{- with $entry.date -}}{{- $dateLabel = time.Format "2 January 2006" . -}}{{- end -}} {{- $id := strings.TrimSuffix (path.Ext $entry.src) $entry.src | urlize -}} {{- $photos = $photos | append (dict "id" $id "title" $entry.title "date" $entry.date "dateLabel" $dateLabel "sortDate" (printf "%s|%s" (default "0000" $entry.date) $entry.src) "story" ($entry.story | default "" | markdownify | string) "preview" (printf "data:%s;base64,%s" $preview.MediaType.Type ($preview.Content | base64Encode)) "thumbnail" $thumbnail.RelPermalink "image" .RelPermalink "width" .Width "height" .Height ) -}} {{- else -}} {{- errorf "Photo album image not found: %s" $entry.src -}} {{- end -}} {{- end -}} {{- return (sort $photos "sortDate" "desc") -}}
